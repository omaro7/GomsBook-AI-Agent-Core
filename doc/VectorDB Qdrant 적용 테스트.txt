4.Qdrant 검색
 4-1) 특정문서 찾기

$body = @{
    limit = 20
    with_payload = $true
    with_vector = $false
    filter = @{
        must = @(
            @{
                key = "projectId"
                match = @{
                    value = "lunchwork_seoul"
                }
            },
            @{
                key = "sourcePath"
                match = @{
                    value = "OEBPS/Text/chapter10_4.xhtml"
                }
            }
        )
    }
} | ConvertTo-Json -Depth 10

$response = Invoke-RestMethod `
    -Method Post `
    -Uri "http://localhost:6333/collections/gomsbook_rag/points/scroll" `
    -ContentType "application/json" `
    -Body $body

$response.result.points | ForEach-Object {
    [PSCustomObject]@{
        id         = $_.id
        projectId  = $_.payload.projectId
        sourcePath = $_.payload.sourcePath
        model      = $_.payload.model
        elementId  = $_.payload.elementId
        content    = $_.payload.content
    }
} | Format-List


 4-2)파일의 Chunk 중 content에서 다음 문장
     낮에는 공부를 하고, 시간이 날 때마다 무료 공연이나 전시를 찾아다녔다.

$response.result.points |
    Where-Object {
        $_.payload.content -like "*무료 공연이나 전시*"
    } |
    ForEach-Object {
        [PSCustomObject]@{
            id         = $_.id
            projectId  = $_.payload.projectId
            sourcePath = $_.payload.sourcePath
            elementId  = $_.payload.elementId
            content    = $_.payload.content
        }
    } | Format-List
