중요한 계산이 하나 있습니다. V3에서는 33 HIT + 1 MISS였고, V4에서는 총 29 HIT입니다. 
그런데 기존 MISS였던 RAG-GOLD-008이 새로 HIT가 됐으므로, 기존 V3의 HIT 중 5건이 HIT → MISS로 회귀했습니다



Windows PowerShell에서 아래를 실행하면 HIT_TO_MISS 5건 + MISS_TO_HIT 1건을 바로 뽑을 수 있습니다.

$v3Path = ".\hybrid-v3\rag-lunchwork_seoul-hybrid-v3-report.json"
$v4Path = ".\hybrid-v4\rag-lunchwork_seoul-hybrid-v4-report.json"

$v3 = Get-Content $v3Path -Raw -Encoding UTF8 | ConvertFrom-Json
$v4 = Get-Content $v4Path -Raw -Encoding UTF8 | ConvertFrom-Json

$v3Map = @{}
foreach ($entry in $v3.entries) {
    $v3Map[$entry.caseId] = $entry
}

$v4Map = @{}
foreach ($entry in $v4.entries) {
    $v4Map[$entry.caseId] = $entry
}

function Get-ExpectedRank($entry) {

    if ($null -eq $entry -or
        $null -eq $entry.retrievedDocuments -or
        $null -eq $entry.expectedDocuments) {
        return $null
    }

    foreach ($document in $entry.retrievedDocuments) {

        foreach ($expected in $entry.expectedDocuments) {

            if ($document.chunkId -eq $expected -or
                $document.sourcePath -eq $expected) {
                return $document.rank
            }

            if ($null -ne $document.chunkId -and
                $document.chunkId.StartsWith($expected + "#")) {
                return $document.rank
            }
        }
    }

    return $null
}

$comparison = foreach ($caseId in $v3Map.Keys | Sort-Object) {

    if (-not $v4Map.ContainsKey($caseId)) {
        continue
    }

    $v3Entry = $v3Map[$caseId]
    $v4Entry = $v4Map[$caseId]

    $v3Hit = [bool]$v3Entry.retrieval.hitAtK
    $v4Hit = [bool]$v4Entry.retrieval.hitAtK

    $status =
        if ($v3Hit -and -not $v4Hit) {
            "HIT_TO_MISS"
        }
        elseif (-not $v3Hit -and $v4Hit) {
            "MISS_TO_HIT"
        }
        elseif ($v3Hit -and $v4Hit) {
            "STABLE_HIT"
        }
        else {
            "STABLE_MISS"
        }

    [PSCustomObject]@{
        caseId            = $caseId
        status            = $status
        question          = $v3Entry.question
        expectedDocuments = ($v3Entry.expectedDocuments -join ", ")
        v3Hit             = $v3Hit
        v4Hit             = $v4Hit
        v3Rank            = Get-ExpectedRank $v3Entry
        v4Rank            = Get-ExpectedRank $v4Entry
        v3Mrr             = $v3Entry.retrieval.mrr
        v4Mrr             = $v4Entry.retrieval.mrr
    }
}

"`n=== HIT_TO_MISS ==="
$comparison |
    Where-Object status -eq "HIT_TO_MISS" |
    Format-Table caseId, v3Rank, v4Rank, v3Mrr, v4Mrr, question -AutoSize

"`n=== MISS_TO_HIT ==="
$comparison |
    Where-Object status -eq "MISS_TO_HIT" |
    Format-Table caseId, v3Rank, v4Rank, v3Mrr, v4Mrr, question -AutoSize
